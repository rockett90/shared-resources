\nomenclature{dBFS}{Decibels relative to the full-scale input of the converter}
\nomenclature{GBW}{Gain-bandwidth product of an operational amplifier}
\nomenclature{LSB}{Least significant bit, the smallest step the converter can resolve}
\nomenclature{$f_0$}{Natural frequency of the filter}
\nomenclature{$f_s$}{Sample rate}
\nomenclature{$Q$}{Quality factor of the filter}
