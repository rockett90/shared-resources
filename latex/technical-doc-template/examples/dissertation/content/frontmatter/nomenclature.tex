\makenomenclature

\nomenclature{ADC}{Analogue-to-Digital Converter}
\nomenclature{ENOB}{Effective number of bits}
\nomenclature{LSB}{Least significant bit, the smallest step the converter can resolve}
\nomenclature{OSR}{Oversampling ratio, the number of conversions averaged into each output sample}
\nomenclature{SNR}{Signal-to-noise ratio}
\nomenclature{$M$}{Oversampling ratio, as used in equations}
\nomenclature{$N$}{Resolution of the converter, in bits}
\nomenclature{$q$}{Size of one quantisation step}
\nomenclature{$\sigma_n$}{Root-mean-square noise at the converter input}
