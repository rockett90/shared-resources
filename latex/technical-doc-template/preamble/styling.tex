% ====================================================================

% --------------------------------------------------------------------
% Visual styling: colours, link appearance, tables, code style, etc.
% ====================================================================

% --------------------------------------------------
% Hyperref link styling and PDF document properties
% --------------------------------------------------
% Requires: \documentTitle and \authorName from metadata.tex
\hypersetup{
  colorlinks=true,
  linkcolor=linkblue,
  citecolor=linkblue,
  urlcolor=linkblue,
  pdftitle={\documentTitle},
  pdfauthor={\authorName}
}

% --------------------------------------------------
% Table styling helpers (optional)
% --------------------------------------------------
\newcommand*{\toprulethick}{\specialrule{1.2pt}{0pt}{3pt}}
\newcommand*{\midrulethin}{\specialrule{0.5pt}{3pt}{3pt}}
\newcommand*{\bottomrulethick}{\specialrule{1.2pt}{3pt}{0pt}}

% --------------------------------------------------
% Colour palette
% --------------------------------------------------
\definecolor{linkblue}{HTML}{005F9E}
\definecolor{darkgrey}{gray}{0.2}
\definecolor{lightgrey}{gray}{0.9}

% Code colours 
\definecolor{codegreen}{rgb}{0,0.6,0}
\definecolor{codegray}{rgb}{0.5,0.5,0.5}
\definecolor{codepurple}{rgb}{0.58,0,0.82}
\definecolor{backcolour}{rgb}{0.95,0.95,0.92}

% --------------------------------------------------
% Code listing style (listings)
% --------------------------------------------------
% Requires: \usepackage{listings} and \usepackage{xcolor} in packages.tex

\lstdefinestyle{mystyle}{
  backgroundcolor=\color{yellow!12},
  commentstyle=\color{codegreen},
  keywordstyle=\color{magenta},
  numberstyle=\footnotesize\itshape\color{codegray},
  stringstyle=\color{codepurple},
  basicstyle=\ttfamily\footnotesize,
  breakatwhitespace=false,
  breaklines=true,
  captionpos=b,
  keepspaces=true,
  numbers=left,
  numbersep=5pt,
  showspaces=false,
  showstringspaces=false,
  showtabs=false,
  tabsize=2,
  frame=lines
}

\lstset{style=mystyle}
