\definecolor{darkgrey}{gray}{0.25}      % VIS-4  secondary text (lightened slightly for cream bg)
\definecolor{lightgrey}{gray}{0.9}      % VIS-4  faint structural rule
\definecolor{creampage}{HTML}{FFFBF4}   % warm cream background (lightened) — easy on the eyes

% ---- Section heading: Libertine, small caps, linkblue + single 1pt rule -----
%      (VIS-5/VIS-7; the 1pt+4pt double rule is reserved for the masthead)
\titleformat{\section}
  {\Large\scshape\color{inknavy}}        % font: large, small caps, blue
  {}{0pt}
  {}
  [\vspace{2pt}{\color{ochre}\titlerule[0.5pt]}\vspace{1.5pt}{\color{linkblue}\titlerule[2pt]}]   % two-tone: warm hairline + blue bar
\titlespacing*{\section}{0pt}{7pt}{6pt}

% ---- Mono accent helper (Inconsolata, dark grey) ----------------------------
\newcommand{\mono}[1]{{\ttfamily\color{darkgrey}#1}}   % VIS-2

% ---- Contact icon: shows the glyph but extracts as nothing (ATS-clean) ------
\newcommand{\cvicon}[1]{\BeginAccSupp{ActualText={}}{\color{linkblue}#1}\EndAccSupp{}}

% ---- Role entry (two-line header + thin underrule — LAY-6) ------------------
%  \role{title}{dates}{company}{location}
\newcommand{\role}[4]{%
  \vspace{2pt}%
  {\large\bfseries\color{linkblue} #1}\hfill{\mono{#2}}\par
  \vspace{-2pt}{\itshape\color{inknavy} #3}\hfill\mono{#4}\par
  \vspace{1.5pt}{\color{lightgrey}\hrule height 0.4pt}\vspace{2pt}%
}

% ---- Short, compact role (CON-8 — early roles) ------------------------------
%  \shortrole{title}{company}{dates}{location}{descriptor}
%  Reuses the two-line header for visual consistency, then one compact descriptor.
\newcommand{\shortrole}[5]{%
  \role{#1}{#3}{#2}{#4}%
  {\color{darkgrey}#5}\par
  \vspace{1pt}%
}

% ---- Education: full-line entry with ochre bullet, styled to match \role ----
%  \edu{qualification}{institution}{dates}
\newcommand{\edu}[3]{%
  \vspace{1.5pt}%
  \noindent
  \makebox[1.2em][l]{\textcolor{ochre}{\textbullet}}%
  \parbox[t]{\dimexpr\linewidth-1.2em-4cm\relax}{\raggedright\strut\bfseries\color{linkblue} #1}%
  \hfill\parbox[t]{4cm}{\raggedleft\strut\mono{#3}}\par
  \vspace{-2pt}\hspace{1.2em}{\itshape\color{inknavy} #2}\par
  \vspace{0.5pt}%
}

% ---- Education: grouped sub-entry beneath an umbrella (CON-4) ---------------
%  \edusub{qualification}{institution}{dates}  — inline date keeps grouped items tidy
\newcommand{\edusub}[3]{%
  \hangindent=1.4em\hangafter=1
  \hspace{1.2em}\textcolor{ochre}{\textbullet}~ #1, \textit{#2} \mono{\textperiodcentered\ #3}\par
}

% ---- Skills row: mono+blue label, then plain text list (ATS-5 — not tabular)
%  \skillrow{label}{contents}
\newcommand{\skillrow}[2]{%
  \mbox{\ttfamily\bfseries\color{linkblue}#1:}\hspace{0.4em}#2\par\vspace{1pt}%
}

% ---- [OPTIONAL] placeholder for a metric absent from the source (CON-3) -----
%  Renders visibly in draft, vanishes when \draftmodefalse. Never an invented number.
\newcommand{\addfigure}[1]{\ifdraftmode\textcolor{red}{\textbf{[\,add figure: #1\,]}}\fi}

% ---- Tight bullet list ------------------------------------------------------
\newlist{cvitems}{itemize}{1}
\setlist[cvitems]{
  leftmargin=1.2em, labelsep=0.5em, topsep=1pt, partopsep=0pt,
  parsep=0pt, itemsep=1pt, label=\textcolor{ochre}{\textbullet}
}
